```js
const css = readFileSync(new URL("../src/styles/tokens.css", import.meta.url), "utf8");
const cor = (nome) => {
  const m = css.match(new RegExp(`--${nome}:\\s*(#[0-9a-fA-F]{6})`));

const lum = (hex) => {
  const c = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255)

  return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2];
};
```

`src/dominio/whatsapp.js`:

```js
// Link do WhatsApp com a mensagem pronta.
export function linkWhatsApp(numero, mensagem) {
  const digitos = String(numero).replace(/\D/g, "");
  if (!digitos) throw new Error("número vazio");
  const base = `https://wa.me/${digitos}`;
  return mensagem ? `${base}?text=${encodeURIComponent(mensagem)}` : base;
}
```

```js
export const formatarPreco = (centavos) => fmt.format(centavos / 100).replace(/\s/g, " ");
```

```js
// Conta quantas respostas apontam para cada lote. Empate: vence o lote que apareceu primeiro nas respostas.
export function resultadoQuiz(perguntas, respostas) {
  const ordem = [];
  const votos = new Map();
  for (const p of perguntas) {
    const op = p.opcoes.find((o) => o.id === respostas?.[p.id]);
    if (!op) return null;
    if (!votos.has(op.lote)) ordem.push(op.lote);
    votos.set(op.lote, (votos.get(op.lote) || 0) + 1);
  }
  let melhor = null;
  for (const lote of ordem) if (melhor === null || votos.get(lote) > votos.get(melhor)) melhor = lote;
  return melhor;
}
```

```js
// Em que etapa do preparo estamos, dado o tempo decorrido.
export const totalSegundos = (etapas) => etapas.reduce((s, e) => s + e.segundos, 0);

export function estadoCronometro(etapas, decorridoMs) {
  const total = totalSegundos(etapas);
  const t = Math.max(0, decorridoMs) / 1000;
  if (t >= total) {
    const u = etapas.length - 1;
    return { indice: u, nome: etapas[u].nome, restanteEtapa: 0, progressoEtapa: 1, progresso: 1, fim: true };
  }
  let inicio = 0;
  for (let i = 0; i < etapas.length; i++) {
    const fimEtapa = inicio + etapas[i].segundos;
    if (t < fimEtapa) {
      const dentro = t - inicio;
      return {
        indice: i, nome: etapas[i].nome,
        restanteEtapa: Math.ceil(etapas[i].segundos - dentro),
        progressoEtapa: dentro / etapas[i].segundos,
        progresso: t / total, fim: false,
      };
    }
    inicio = fimEtapa;
  }
}

export const formatarTempo = (s) => `${Math.floor(s / 60)}:${String(Math.floor(s % 60)).padStart(2, "0")}`;
```

```js
// Revela elementos ao entrar na tela: sobe e aparece, conta números e desenha traços SVG.
import { valorContagem } from "../dominio/contagem.js";

const reduz = () => matchMedia("(prefers-reduced-motion: reduce)").matches;

function contar(el) {
  const alvo = Number(el.dataset.contar);
  if (reduz()) { el.textContent = alvo.toLocaleString("pt-BR"); return; }
  const t0 = performance.now();
  const passo = (agora) => {
    const t = (agora - t0) / 1400;
    el.textContent = valorContagem(alvo, t).toLocaleString("pt-BR");
    if (t < 1) requestAnimationFrame(passo);
  };
  requestAnimationFrame(passo);
}

export function iniciarRevelar(raiz = document) {
  const alvos = raiz.querySelectorAll("[data-revelar], [data-contar], [data-desenhar]");
  if (!("IntersectionObserver" in window) || reduz()) {
    alvos.forEach((el) => { el.classList.add("visivel"); if (el.dataset.contar) contar(el); });
    return;
  }
  const io = new IntersectionObserver((entradas) => {
    for (const e of entradas) {
      if (!e.isIntersecting) continue;
      e.target.classList.add("visivel");
      if (e.target.dataset.contar) contar(e.target);
      io.unobserve(e.target);
    }
  }, { rootMargin: "0px 0px -12% 0px" });
  alvos.forEach((el) => io.observe(el));
}
```

```jsx
import { useRef, useState } from "react";
import { CARDAPIO } from "../dados/cardapio.js";
import { formatarPreco } from "../dominio/preco.js";

export default function Cardapio() {
  const [ativa, setAtiva] = useState(0);
  const abas = useRef([]);
  const teclas = (e) => {
    const d = e.key === "ArrowRight" ? 1 : e.key === "ArrowLeft" ? -1 : 0;
    if (!d) return;
    const n = (ativa + d + CARDAPIO.length) % CARDAPIO.length;
    setAtiva(n);
    abas.current[n]?.focus();
  };
  const cat = CARDAPIO[ativa];
  return (
    <div className="cardapio">
      <div role="tablist" aria-label="Categorias do cardápio" className="abas" onKeyDown={teclas}>
        {CARDAPIO.map((c, i) => (
          <button key={c.id} ref={(el) => (abas.current[i] = el)} role="tab" id={`aba-${c.id}`} aria-controls={`painel-${c.id}`}
            aria-selected={i === ativa} tabIndex={i === ativa ? 0 : -1} className="aba" onClick={() => setAtiva(i)}>
            {c.nome}
          </button>
        ))}
      </div>
      <ul key={cat.id} role="tabpanel" id={`painel-${cat.id}`} aria-labelledby={`aba-${cat.id}`} className="itens">
        {cat.itens.map((it) => (
          <li key={it.nome} className="item">
            <div>
              <h3>{it.nome}</h3>
              <p className="desc">{it.descricao}</p>
              {it.notas && <p className="notas">{it.notas.join(" · ")}</p>}
            </div>
            <span className="preco">{formatarPreco(it.preco)}</span>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

### Task 9: Guia de preparo com cronômetro

**Files:**
- Create: `src/secoes/Preparo.jsx`
- Modify: `src/styles/secoes.css`, `src/pages/index.astro`

**Interfaces:**
- Consumes: `METODOS`, `estadoCronometro`, `formatarTempo`, `totalSegundos`.
- Produces: `<Preparo client:visible />`; botões com nomes acessíveis "Iniciar", "Pausar", "Zerar"; tempo em `[data-tempo]`, etapa em `[data-etapa]` (usados no e2e).

- [ ] **Step 1: Componente**

`src/secoes/Preparo.jsx`:
```jsx
import { useEffect, useRef, useState } from "react";
import { METODOS } from "../dados/preparo.js";
import { estadoCronometro, formatarTempo, totalSegundos } from "../dominio/cronometro.js";

const R = 54, C = 2 * Math.PI * R;

export default function Preparo() {
  const [mi, setMi] = useState(0);
  const [rodando, setRodando] = useState(false);
  const [ms, setMs] = useState(0);
  const acumulado = useRef(0);   // ms já contados antes da última partida
  const partida = useRef(0);     // performance.now() da última partida
  const zerou = useRef(false);   // Zerar com o relógio andando: a limpeza do efeito não deve somar tempo
  const raf = useRef(0);
  const m = METODOS[mi];
  const st = estadoCronometro(m.etapas, ms);

  // Um único laço por vez: o efeito depende só de "rodando" e se limpa sozinho.
  useEffect(() => {
    if (!rodando) return;
    partida.current = performance.now();
    const passo = (agora) => {
      const t = acumulado.current + (agora - partida.current);
      setMs(t);
      if (estadoCronometro(m.etapas, t).fim) { setRodando(false); return; }
      raf.current = requestAnimationFrame(passo);
    };
    raf.current = requestAnimationFrame(passo);
    return () => {
      cancelAnimationFrame(raf.current);
      if (zerou.current) zerou.current = false;
      else acumulado.current += performance.now() - partida.current;
    };
  }, [rodando, m.etapas]);

  const zerar = () => { if (rodando) zerou.current = true; setRodando(false); acumulado.current = 0; setMs(0); };
  const trocar = (i) => { zerar(); setMi(i); };
  const restanteTotal = Math.max(0, Math.ceil(totalSegundos(m.etapas) - ms / 1000));

  return (
    <div className="preparo">
      <div className="abas" role="group" aria-label="Método de preparo">
        {METODOS.map((x, i) => (
          <button key={x.id} className="aba" aria-pressed={i === mi} onClick={() => trocar(i)}>{x.nome}</button>
        ))}
      </div>
      <div className="preparo-grade">
        <dl className="ficha">
          <dt>Proporção</dt><dd>{m.proporcao}</dd>
          <dt>Dose</dt><dd>{m.dose}</dd>
          <dt>Água</dt><dd>{m.agua}</dd>
          <dt>Moagem</dt><dd>{m.moagem}</dd>
          <dt>Temperatura</dt><dd>{m.temperatura}</dd>
        </dl>
        <div className="relogio">
          <svg viewBox="0 0 120 120" aria-hidden="true">
            <circle cx="60" cy="60" r={R} fill="none" stroke="var(--bege)" strokeWidth="8" />
            <circle cx="60" cy="60" r={R} fill="none" stroke="var(--caramelo)" strokeWidth="8" strokeLinecap="round"
              strokeDasharray={C} strokeDashoffset={C * (1 - st.progresso)} transform="rotate(-90 60 60)" />
          </svg>
          <p className="tempo preco" data-tempo aria-live="off">{formatarTempo(restanteTotal)}</p>
          <p className="etapa" data-etapa aria-live="polite">{st.fim ? "Pronto. Bom café!" : `${st.indice + 1}/${m.etapas.length} · ${st.nome}`}</p>
          <div className="controles">
            {rodando
              ? <button className="botao" onClick={() => setRodando(false)}>Pausar</button>
              : <button className="botao" onClick={() => { if (st.fim) zerar(); setRodando(true); }}>Iniciar</button>}
            <button className="botao contorno" onClick={zerar}>Zerar</button>
          </div>
        </div>
      </div>
    </div>
  );
}
```

Nota de comportamento: clicar "Iniciar" várias vezes não cria laços extras — depois do primeiro clique o botão vira "Pausar", e `setRodando(true)` repetido não reexecuta o efeito. Iniciar depois de "Pronto" recomeça do zero. "Zerar" com o relógio andando marca `zerou`, e a limpeza do efeito (que roda depois) não soma o tempo de volta — sem isso, o próximo "Iniciar" começaria adiantado.

```jsx
import { useState } from "react";
import { PERGUNTAS } from "../dados/quiz.js";
import { LOTES } from "../dados/lotes.js";
import { resultadoQuiz } from "../dominio/quiz.js";
import simbolo from "../assets/logo/ls-brew-simbolo-fundo-claro.svg?url";

export default function Quiz() {
  const [passo, setPasso] = useState(0);
  const [resp, setResp] = useState({});
  const [erro, setErro] = useState("");
  const [lote, setLote] = useState(null);
  const p = PERGUNTAS[passo];

  const avancar = () => {
    if (!resp[p.id]) { setErro("Escolha uma opção"); return; }
    setErro("");
    if (passo < PERGUNTAS.length - 1) setPasso(passo + 1);
    else setLote(LOTES.find((l) => l.id === resultadoQuiz(PERGUNTAS, resp)));
  };
  const refazer = () => { setPasso(0); setResp({}); setLote(null); setErro(""); };

  if (lote) return (
    <div className="quiz-resultado" aria-live="polite">
      <img src={simbolo} alt="" width="72" height="72" />
      <p className="rotulo">Seu café é</p>
      <h3>{lote.nome}</h3>
      <p>{lote.descricao}</p>
      <p className="notas">{lote.notas.join(" · ")}</p>
      <div className="controles"><a className="botao" href="#cardapio">Ver no cardápio</a><button className="botao contorno" onClick={refazer}>Refazer</button></div>
    </div>
  );

  return (
    <fieldset key={p.id} className="quiz-pergunta">
      <legend><span className="rotulo">Pergunta {passo + 1} de {PERGUNTAS.length}</span><h3>{p.texto}</h3></legend>
      <div className="opcoes">
        {p.opcoes.map((o) => (
          <button key={o.id} type="button" className="opcao" aria-pressed={resp[p.id] === o.id}
            onClick={() => { setResp({ ...resp, [p.id]: o.id }); setErro(""); }}>{o.texto}</button>
        ))}
      </div>
      {erro && <p role="alert" className="erro">{erro}</p>}
      <button type="button" className="botao" onClick={avancar}>{passo < PERGUNTAS.length - 1 ? "Próxima" : "Ver resultado"}</button>
    </fieldset>
  );
}
```

```js
import { deveMostrarEntrada } from "../dominio/entrada.js";

// Mostra o grão se montando uma vez por visita; sai quando as fontes chegam (mínimo 1,2 s, máximo 6 s).
export async function iniciarEntrada() {
  const el = document.getElementById("entrada");
  if (!el) return;
  let visto = false;
  try { visto = sessionStorage.getItem("ls-brew:entrada") === "1"; } catch {}
  const forcar = new URLSearchParams(location.search).get("abertura") === "1";
  if (!deveMostrarEntrada({ visto, automatizado: !!navigator.webdriver, forcar })) { el.remove(); dispatchEvent(new Event("ls-brew:entrada-fim")); return; }
  try { sessionStorage.setItem("ls-brew:entrada", "1"); } catch {}
  const minimo = matchMedia("(prefers-reduced-motion: reduce)").matches ? 400 : 1200;
  await Promise.race([Promise.all([document.fonts?.ready, new Promise((r) => setTimeout(r, minimo))]), new Promise((r) => setTimeout(r, 6000))]);
  el.classList.add("saindo");
  dispatchEvent(new Event("ls-brew:entrada-fim"));
  setTimeout(() => el.remove(), 520);
}
```

```jsx
import { useEffect, useState } from "react";
import ScrollSequence from "./ScrollSequence.jsx";
import { faseDaAbertura, FRASES } from "../dominio/abertura.js";

export default function AberturaCena({ base, quadros, logo }) {
  const [fase, setFase] = useState(0);
  const [reduz, setReduz] = useState(false);
  useEffect(() => setReduz(matchMedia("(prefers-reduced-motion: reduce)").matches), []);
  return (
    <ScrollSequence pasta={`${base}seq/v60`} pastaCelular={`${base}seq/v60-celular`} quadros={quadros}

      alt="Café coado na V60 caindo na xícara" className="abertura" onProgresso={(p) => setFase(faseDaAbertura(p))}>
      <div className="abertura-texto">
        {reduz ? (
          <>{FRASES.map((f) => <p key={f} className="frase visivel">{f}</p>)}<img className="abertura-logo visivel" src={logo} alt="LS Brew" /></>
        ) : (
          <>
            {FRASES.map((f, i) => <p key={f} className={`frase ${fase === i ? "visivel" : ""}`} aria-hidden={fase !== i}>{f}</p>)}
            <img className={`abertura-logo ${fase === 3 ? "visivel" : ""}`} src={logo} alt="LS Brew" />
          </>
        )}
        <h1 className="sr">LS Brew · café especial da Serra de Baturité em Fortaleza</h1>
      </div>
    </ScrollSequence>
  );
}
```

```js
// Roteiro da página inteira: rode depois de "npm run build" com "node e2e/pagina.mjs".
import { servidor, navegador, pagina, confere, resultado, URL_BASE } from "./util.mjs";

const parar = await servidor();
const browser = await navegador();
try {
  for (const [nome, opts] of [["computador", {}], ["celular", { viewport: { width: 375, height: 812 }, isMobile: true, hasTouch: true }]]) {
    console.log(`\n${nome}`);
    const { ctx, page, erros } = await pagina(browser, opts);
    await page.goto(URL_BASE);
    confere(await page.locator("#entrada").count() === 0, "entrada do grão não aparece no navegador automatizado");
    confere(await page.locator('meta[name="robots"][content="noindex"]').count() === 1, "noindex presente");
    for (const id of ["origem", "cardapio", "preparo", "quiz", "clube", "galeria", "visite"])
      confere(await page.locator(`#${id}`).count() === 1, `seção #${id}`);

    // Nada estoura a largura
    const largura = await page.evaluate(() => [document.documentElement.scrollWidth, innerWidth]);
    confere(largura[0] <= largura[1], `sem rolagem lateral (${largura[0]} <= ${largura[1]})`);

    // Abertura: rola até o fim da cena e vê o logo
    await page.evaluate(() => scrollTo(0, innerHeight * 2.9));
    await page.waitForTimeout(800);
    confere(await page.locator(".abertura-logo.visivel").count() === 1, "logo aparece no fim da abertura");

    // Cardápio
    await page.locator("#cardapio").scrollIntoViewIfNeeded();
    await page.getByRole("tab", { name: "Gelados" }).click();
    confere(await page.getByRole("tabpanel").getByText("Cold brew").isVisible(), "aba Gelados mostra Cold brew");
    confere(/R\$ \d/.test(await page.getByRole("tabpanel").locator(".preco").first().innerText()), "preço formatado");

    // Cronômetro: Iniciar 3 vezes não acelera
    await page.locator("#preparo").scrollIntoViewIfNeeded();
    await page.getByRole("button", { name: "Espresso", exact: true }).click();
    const iniciar = page.getByRole("button", { name: "Iniciar" });
    await iniciar.click(); await page.getByRole("button", { name: "Pausar" }).click(); await iniciar.click();
    await page.waitForTimeout(2000);
    const tempo = await page.locator("[data-tempo]").innerText();
    confere(tempo === "0:28" || tempo === "0:27", `cronômetro anda 1 s por segundo (${tempo})`);
    await page.getByRole("button", { name: "Zerar" }).click();
    confere(await page.locator("[data-tempo]").innerText() === "0:30", "Zerar volta a 0:30");
    await iniciar.click(); await page.waitForTimeout(1000);
    const depois = await page.locator("[data-tempo]").innerText();
    confere(depois === "0:29" || depois === "0:28", `depois de Zerar, recomeça do zero (${depois})`);
    await page.getByRole("button", { name: "Zerar" }).click();

    // Quiz
    await page.locator("#quiz").scrollIntoViewIfNeeded();
    await page.getByRole("button", { name: "Próxima" }).click();
    confere(await page.getByRole("alert").filter({ hasText: "Escolha uma opção" }).isVisible(), "quiz exige resposta");
    for (const op of ["Cedo e cheia de energia", "Torta de limão", "Coado, puro, sem açúcar"]) {
      await page.getByRole("button", { name: op }).click(); await page.getByRole("button", { name: "Próxima" }).click();
    }
    await page.getByRole("button", { name: "Sítio na serra, rede e fogão a lenha" }).click();
    await page.getByRole("button", { name: "Ver resultado" }).click();
    confere(await page.getByRole("heading", { name: "Pedra Branca" }).isVisible(), "quiz b,b,b,a → Pedra Branca");

    // WhatsApp
    const href = await page.getByRole("link", { name: "Pedir pelo WhatsApp" }).getAttribute("href");
    confere(href.startsWith("https://wa.me/5585000000000?text=Ol%C3%A1"), "link do WhatsApp com mensagem");

    // Newsletter
    const campo = page.getByLabel("Seu e-mail");
    await campo.fill("ana"); await page.getByRole("button", { name: "Assinar" }).click();
    confere(await page.getByRole("alert").filter({ hasText: "e-mail válido" }).isVisible(), "newsletter recusa e-mail inválido");
    await campo.fill(" Ana@Ex.com "); await page.getByRole("button", { name: "Assinar" }).click();
    confere(await page.getByText("Inscrito!").isVisible(), "newsletter aceita e agradece");

    confere(erros.length === 0, `sem erros no console ${erros.length ? JSON.stringify(erros) : ""}`);
    await page.screenshot({ path: `_shots/${nome}.png`, fullPage: true });
    await ctx.close();
  }

  console.log("\nreduzir movimento");
  const { ctx, page } = await pagina(browser, { reducedMotion: "reduce" });
  await page.goto(URL_BASE);
  await page.waitForFunction(() => document.querySelectorAll(".frase.visivel").length === 3, null, { timeout: 5000 }).catch(() => {});
  confere(await page.locator(".frase.visivel").count() === 3, "as três frases aparecem paradas");
  confere(await page.locator("#origem .lote").first().evaluate((el) => getComputedStyle(el).opacity) === "1", "conteúdo visível sem animação");
  await ctx.close();
} finally {
  await browser.close();
  parar();
}
process.exit(resultado() ? 1 : 0);
```
